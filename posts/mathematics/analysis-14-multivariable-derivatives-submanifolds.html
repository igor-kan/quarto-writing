1. The Conceptual Shift: Beyond Partial Derivatives

Total differentiability requires the existence of a linear operator \(Df(x_0)\) satisfying: \[\lim_{h \to 0} \frac{\|f(x_0 + h) - f(x_0) - Df(x_0)h\|}{\|h\|} = 0\] Surfaces are submanifolds (regular level sets), and Lagrange multipliers span the normal space to the tangent space.

import numpy as np

f = lambda x: x[0]**2 * np.sin(x[1]) + np.exp(x[0]*x[1])
grad_f = lambda x: np.array([2*x[0]*np.sin(x[1]) + x[1]*np.exp(x[0]*x[1]),
                             x[0]**2*np.cos(x[1]) + x[0]*np.exp(x[0]*x[1])])

x0 = np.array([1.5, 0.8])
h = np.array([1e-5, -1e-5])
residual = abs(f(x0 + h) - f(x0) - np.dot(grad_f(x0), h))
print(f"Python 3: Fréchet Residual / ||h|| = {residual / np.linalg.norm(h):.4e}")
#include <iostream>
#include <cmath>

int main() {
    auto f = [](double x, double y) { return x*x*std::sin(y) + std::exp(x*y); };
    double x = 1.5, y = 0.8;
    double gx = 2*x*std::sin(y) + y*std::exp(x*y);
    double gy = x*x*std::cos(y) + x*std::exp(x*y);

    double hx = 1e-5, hy = -1e-5;
    double h_norm = std::hypot(hx, hy);
    double res = std::abs(f(x + hx, y + hy) - f(x, y) - (gx * hx + gy * hy));
    std::cout << "C++20:    Fréchet Residual / ||h|| = " << (res / h_norm) << "\n";
    return 0;
}

program frechet_check
  implicit none
  real(8) :: x, y, gx, gy, hx, hy, h_norm, f0, fh, res

  x = 1.5d0; y = 0.8d0
  f0 = x*x*sin(y) + exp(x*y)
  gx = 2.0d0*x*sin(y) + y*exp(x*y)
  gy = x*x*cos(y) + x*exp(x*y)

  hx = 1.0d-5; hy = -1.0d-5
  h_norm = sqrt(hx*hx + hy*hy)
  fh = (x+hx)**2 * sin(y+hy) + exp((x+hx)*(y+hy))
  res = abs(fh - f0 - (gx*hx + gy*hy))

  print '(A,E12.4)', "Fortran:  Residual / ||h|| = ", (res / h_norm)
end program frechet_check
